Demo JS nhỏ: đổi chữ khi click

Chưa bấm.

const btn = document.querySelector('#demoBtn');
btn.addEventListener('click', () => {
  document.querySelector('#demoOut').textContent = 'Đã bấm ' + new Date().toLocaleTimeString();
});
